cmake_minimum_required(VERSION 3.24)
project(InstalledConsumer LANGUAGES CXX)
if(STATIC_ZLIB)
  # In the system variant, resolve the package's zlib dependency to the same
  # explicitly selected header/archive as the producing build and C control.
  set(ZLIB_LIBRARY "${STATIC_ZLIB}")
  set(ZLIB_INCLUDE_DIR "${STATIC_ZLIB_INCLUDE}")
endif()
find_package(GhidraDecompiler REQUIRED CONFIG)
enable_testing()
add_executable(capabilities decomp_smoke.cc)
target_link_libraries(capabilities PRIVATE ghidra::decomp)
add_test(NAME capabilities COMMAND capabilities --check-capabilities)
if(TEST_TOY)
  foreach(file toy.ldefs toy_builder_be_align2.sla toy.pspec toy.cspec)
    if(NOT EXISTS "${GhidraDecompiler_SLEIGH_HOME}/Ghidra/Processors/Toy/data/languages/${file}")
      message(FATAL_ERROR "Selected installed Toy language missing ${file}")
    endif()
  endforeach()
  add_executable(toy_decompile toy_decompile.cc)
  target_link_libraries(toy_decompile PRIVATE ghidra::decomp)
  if(STATIC_ZLIB)
    if(NOT EXISTS "${STATIC_ZLIB}" OR NOT EXISTS "${STATIC_ZLIB_INCLUDE}/zlib.h")
      message(FATAL_ERROR "External static gzip archive/header unavailable")
    endif()
    file(READ "${STATIC_ZLIB}" archive_magic LIMIT 8 HEX)
    if(NOT archive_magic STREQUAL "213c617263683e0a" AND NOT archive_magic STREQUAL "213c7468696e3e0a")
      message(FATAL_ERROR "External gzip control must be a static archive")
    endif()
    enable_language(C)
    add_library(external_gzip OBJECT external_gzip.c)
    target_include_directories(external_gzip PRIVATE "${STATIC_ZLIB_INCLUDE}")
    target_sources(toy_decompile PRIVATE $<TARGET_OBJECTS:external_gzip>)
    target_compile_definitions(toy_decompile PRIVATE GHIDRA_TEST_GZIP)
    target_link_libraries(toy_decompile PRIVATE "${STATIC_ZLIB}")
  else()
    message(STATUS "External static gzip co-link control unavailable: STATIC_ZLIB not supplied")
  endif()
  add_test(NAME toy_decompile COMMAND toy_decompile "${GhidraDecompiler_SLEIGH_HOME}" "${CMAKE_CURRENT_SOURCE_DIR}/skipnext2-image.xml")
else()
  message(STATUS "Installed Toy decompilation unavailable: Toy intentionally not selected")
endif()
